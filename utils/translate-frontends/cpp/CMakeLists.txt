cmake_minimum_required(VERSION 3.20)
project(NeverCCppFrontend LANGUAGES C CXX)
find_package(LLVM 20.1.8 EXACT REQUIRED CONFIG)
find_package(Clang 20.1.8 EXACT REQUIRED CONFIG)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL "20.1.8")
  message(FATAL_ERROR "neverc-cpp-frontend requires LLVM/Clang 20.1.8")
endif()
add_executable(neverc-cpp-frontend Frontend.cpp Lowering.cpp Project.cpp MathSDK.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  Frontend.cpp Lowering.cpp Project.cpp MathSDK.cpp Frontend.h sdk/approved-sdk.json)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/Frontend.cpp" _frontend_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/Lowering.cpp" _lowering_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/Project.cpp" _project_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/MathSDK.cpp" _math_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/SDKCatalog.h.in" _sdk_template_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/BuildID.h.in" _build_template_hash)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk/approved-sdk.json" NEVERC_CPP_SDK_CATALOG)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/sdk/approved-sdk.json" NEVERC_CPP_SDK_CATALOG_HASH)
configure_file(SDKCatalog.h.in SDKCatalog.h @ONLY)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/Frontend.h" _header_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" _cmake_hash)
string(SHA256 NEVERC_CPP_BUILD_ID
  "20.1.8:${_frontend_hash}:${_lowering_hash}:${_project_hash}:${_math_hash}:${_header_hash}:${_cmake_hash}:${_sdk_template_hash}:${_build_template_hash}:${NEVERC_CPP_SDK_CATALOG_HASH}")
configure_file(BuildID.h.in BuildID.h @ONLY)
set_target_properties(neverc-cpp-frontend PROPERTIES
  CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(neverc-cpp-frontend SYSTEM PRIVATE
  ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_include_directories(neverc-cpp-frontend PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(neverc-cpp-frontend PRIVATE ${LLVM_DEFINITIONS})
target_link_libraries(neverc-cpp-frontend PRIVATE clang-cpp LLVM)
install(TARGETS neverc-cpp-frontend RUNTIME DESTINATION bin)
install(FILES sdk/neverc-cpp-sdk.example.json DESTINATION share/neverc)
